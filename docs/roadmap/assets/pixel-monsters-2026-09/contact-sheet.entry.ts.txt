import { SPRITES, spriteFor } from '/home/user/deck-monsters/apps/web/src/animations/pixel-fight/sprites.ts';
import { paletteFor } from '/home/user/deck-monsters/apps/web/src/animations/pixel-fight/appearance-palette.ts';
import { drawSprite } from '/home/user/deck-monsters/apps/web/src/animations/pixel-fight/renderer.ts';

const SCALE = 3;
const THEMES: Array<[string, string, string]> = [
  ['Phosphor', '#0a0e0a', '#a8d060'], ['Amber', '#0a0800', '#ffb000'],
  ['Ember', '#12060a', '#ff6f61'], ['SNES', '#060c1e', '#f8c000'],
];
const LOOKS: Array<[string, string, string]> = [
  ['Basilisk', '', 'Stonefang'],
  ['Basilisk', 'gold and black diamond patterned', 'Stonefang'],
  ['Basilisk', 'crimson scales', 'Stonefang'],
  ['Basilisk', 'green', 'Gin & Tonic'],
  ['Basilisk', 'green', 'Slinky'],
  ['Basilisk', 'jet black', 'Stonefang'],
  ['Gladiator', 'tattered rags', 'Ben Franklin'],
  ['Gladiator', 'royal blue', 'Ben Franklin'],
  ['Minotaur', 'scarred, wrinkled, and beautifully auburn', 'Taurus'],
  ['Jinn', 'slightly translucent blue', 'Djinnjer'],
  ['Weeping Angel', 'deceptively glorious', 'Seraph'],
  ['Weeping Angel', 'pale silver', 'Seraph'],
];

const root = document.getElementById('root')!;
for (const [themeName, bg, fg] of THEMES) {
  const row = document.createElement('div');
  row.style.cssText = `background:${bg};color:${fg};padding:6px 8px;display:flex;gap:10px;align-items:flex-end;font:11px monospace`;
  const label = document.createElement('div');
  label.textContent = themeName; label.style.width = '62px';
  row.appendChild(label);
  for (const [type, look, name] of LOOKS) {
    const cell = document.createElement('div');
    cell.style.cssText = 'display:flex;flex-direction:column;align-items:center;width:78px;text-align:center';
    const canvas = document.createElement('canvas');
    canvas.width = 24 * SCALE; canvas.height = 24 * SCALE;
    const ctx = canvas.getContext('2d')!;
    const sprite = spriteFor(type);
    drawSprite(ctx, sprite.frames.idle[0], paletteFor(sprite.palette, look, name), 0, 0, SCALE, { mirror: false, flash: false });
    cell.appendChild(canvas);
    if (themeName === 'Phosphor') {
      const cap = document.createElement('div');
      cap.textContent = `${type.split(' ')[0]}: ${look || '(none)'}${look === 'green' ? ` · ${name}` : ''}`;
      cap.style.cssText = 'font-size:9px;line-height:1.1;margin-top:3px;opacity:.85';
      cell.appendChild(cap);
    }
    row.appendChild(cell);
  }
  root.appendChild(row);
}
void SPRITES;
